  <script>
    const boton = document.getElementById("startButton");
    const inicio = document.getElementById("inicio");
    const susto = document.getElementById("susto");
    const fondo = document.getElementById("fondo");
    const contador = document.getElementById("contador");

    // Crear el fondo repetido
    for (let i = 0; i < 45; i++) {
      const texto = document.createElement("span");
      texto.textContent = "DIEGO ES GAY";
      fondo.appendChild(texto);
    }

    boton.addEventListener("click", async () => {

      inicio.style.display = "none";
      susto.style.display = "flex";

      // Intentar pantalla completa
      try {
        if (document.documentElement.requestFullscreen) {
          await document.documentElement.requestFullscreen();
        }
      } catch (e) {
        console.log("Pantalla completa no disponible");
      }

      // Vibración si el celular la permite
      if (navigator.vibrate) {
        navigator.vibrate([500, 200, 500, 200, 1000]);
      }

      // Crear sonido tipo bocina usando Web Audio
      const AudioContext =
        window.AudioContext || window.webkitAudioContext;

      if (AudioContext) {
        const audio = new AudioContext();

        const oscillator = audio.createOscillator();
        const gain = audio.createGain();

        oscillator.type = "sawtooth";
        oscillator.frequency.value = 120;

        gain.gain.value = 0.12;

        oscillator.connect(gain);
        gain.connect(audio.destination);

        oscillator.start();

        // Cambiar ligeramente el tono para hacerlo más molesto
        const intervalo = setInterval(() => {
          oscillator.frequency.value =
            100 + Math.random() * 80;
        }, 150);

        setTimeout(() => {
          clearInterval(intervalo);
          oscillator.stop();
          audio.close();
        }, 60000);
      }

      // Contador de 60 segundos
      let segundos = 60;

      const cuenta = setInterval(() => {
        segundos--;

        contador.textContent =
          "Sorpresa... " + segundos + " segundos 😈";

        if (segundos <= 0) {
          clearInterval(cuenta);

          susto.style.display = "none";
          inicio.style.display = "block";

          // Salir de pantalla completa
          if (document.fullscreenElement) {
            document.exitFullscreen().catch(() => {});
          }
        }
      }, 1000);
    });
  </script>
